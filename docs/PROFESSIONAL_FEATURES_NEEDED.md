**Database Schema:**

```sql
CREATE TABLE publications (
  id UUID PRIMARY KEY,
  title TEXT NOT NULL,
  authors JSONB,
  abstract TEXT,
  doi TEXT UNIQUE,
  publication_date DATE,
  journal_name TEXT,
  volume TEXT,
  issue TEXT,
  pages TEXT,
  citation_count INTEGER DEFAULT 0,
  impact_factor DECIMAL(4,2),
  keywords TEXT[],
  pdf_url TEXT,
  open_access BOOLEAN DEFAULT false,
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_publications_date ON publications(publication_date);
CREATE INDEX idx_publications_citations ON publications(citation_count);
CREATE INDEX idx_publications_keywords ON publications USING GIN(keywords);
```

#### **1.2 Bookmarks API** (3 vị trí)

```javascript
// Backend APIs
GET /api/bookmarks
```

**Database Schema:**

```sql
CREATE TABLE bookmarks (
  id UUID PRIMARY KEY,
  user_id UUID REFERENCES users(id),
  bookmark_type VARCHAR(20), -- 'paper', 'journal', 'topic', 'keyword'
  item_id UUID,
  item_data JSONB,
  notes TEXT,
  tags TEXT[],
  created_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_bookmarks_user ON bookmarks(user_id);
CREATE INDEX idx_bookmarks_type ON bookmarks(bookmark_type);
```

**Database Schema:**

```sql
CREATE TABLE researcher_stats (
  id UUID PRIMARY KEY,

  stat_date DATE,
  total_publications INTEGER DEFAULT 0,
  total_citations INTEGER DEFAULT 0,
```

### **4. Pagination Implementation**

#### **4.1 Backend Pagination**

```csharp
public class PaginationParams
{
    public int Page { get; set; } = 1;
    public int Limit { get; set; } = 20;
}

public class PaginatedResult<T>
{
    public List<T> Data { get; set; }
    public int Page { get; set; }
    public int Limit { get; set; }
    public int TotalCount { get; set; }
    public int TotalPages => (int)Math.Ceiling(TotalCount / (double)Limit);
    public bool HasNext => Page < TotalPages;
    public bool HasPrevious => Page > 1;
}

// In controller
[HttpGet]
public async Task<PaginatedResult<Publication>> GetPublications(
    [FromQuery] PaginationParams pagination)
{
    var query = _context.Publications.AsQueryable();
    var total = await query.CountAsync();
    var data = await query
        .Skip((pagination.Page - 1) * pagination.Limit)
        .Take(pagination.Limit)
        .ToListAsync();

    return new PaginatedResult<Publication>
    {
        Data = data,
        Page = pagination.Page,
        Limit = pagination.Limit,
        TotalCount = total
    };
}
```

```javascript
function Pagination({ currentPage, totalPages, onPageChange }) {
  const pages = Array.from({ length: totalPages }, (_, i) => i + 1);

  return (
    <div className="pagination">
      <button
        disabled={currentPage === 1}
        onClick={() => onPageChange(currentPage - 1)}
      >
        Previous
      </button>

      {pages.map((page) => (
        <button
          key={page}
          className={page === currentPage ? "active" : ""}
          onClick={() => onPageChange(page)}
        >
          {page}
        </button>
      ))}

      <button
        disabled={currentPage === totalPages}
        onClick={() => onPageChange(currentPage + 1)}
      >
        Next
      </button>
    </div>
  );
}
```
